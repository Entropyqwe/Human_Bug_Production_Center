cmake_minimum_required(VERSION 3.5)
project(CALC)
set(CMAKE_CXX_STANDARD 11)
set(EXECUTABLE_OUTPUT_PATH ${CMAKE_BINARY_DIR}/bin)
include_directories(${PROJECT_SOURCE_DIR}/include)
file(GLOB LIB_SRC src/add.cpp src/sub.cpp src/mult.cpp src/div.cpp)
set(APP_SRC src/main.cpp)
add_library(lib_static STATIC ${LIB_SRC})
set_target_properties(lib_static PROPERTIES OUTPUT_NAME "y")
add_library(lib_shared SHARED ${LIB_SRC})
set_target_properties(lib_shared PROPERTIES OUTPUT_NAME "y")
add_executable(app ${APP_SRC})
target_link_libraries(app PRIVATE lib_static)
